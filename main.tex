\begin{algorithm}
\caption{HSeeker Pipeline: Order of Stages}
\label{alg:pipeline}
\small
\begin{algorithmic}[1]
\Require Sequence $S$, parameters of Algorithm~\ref{alg:mirror-phase1},
         overlap strategy $\sigma \in \{\textsc{Greedy}, \textsc{Score}\}$,
         flags $\mathit{score}$ (apply Algorithm~\ref{alg:mirror-phase3}),
         $\mathit{hp}$ (homopolymer filter), optional AT threshold $\tau$
         ($\sigma = \textsc{Score}$ requires $\mathit{score}$)
\Ensure Reported hits $\mathcal{H}$
\State $\mathcal{C} \gets \textsc{Phase1}(S)$
    \Comment{enumeration, inward extension, deduplication}
\If{$\sigma = \textsc{Greedy}$}
    \State $\mathcal{C} \gets \textsc{Phase2}(\mathcal{C}, \textsc{Greedy})$
        \Comment{overlaps resolved before any filter or score}
\EndIf
\If{$\tau$ is given}
    \State drop $h \in \mathcal{C}$ whose full motif has A/T fraction $\ge \tau$
        \Comment{arms and spacer both counted}
\EndIf
\If{$\mathit{score}$}
    \State score every $h \in \mathcal{C}$ with Algorithm~\ref{alg:mirror-phase3}
    \If{$\mathit{hp}$}
        \State drop $h$ if its full motif, or its putative triplex with the
               brackets removed, uses a single base
    \EndIf
\ElsIf{$\mathit{hp}$}
    \State drop $h$ if its full motif uses a single base
\EndIf
\If{$\sigma = \textsc{Score}$}
    \State $\mathcal{C} \gets \textsc{Phase2}(\mathcal{C}, \textsc{Score})$
        \Comment{filters and scoring come first}
\EndIf
\State \Return $\mathcal{C}$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}
\caption{HSeeker Phase 1: Candidate Enumeration and Normalization}
\label{alg:mirror-phase1}
\scriptsize
\begin{algorithmic}[1]
\Require
    DNA sequence $S$ of length $n$,
    minimum arm length $L_{\min}$,
    maximum arm length $L_{\max}$,
    maximum spacer $S_{\max}$,
    purity threshold $P \in [0,1]$,
    mismatch tolerance $M \in [0,1]$,
    flag $\mathit{rmq}$ (enable the purity RMQ prefilter)
\Ensure Candidate mirror repeat list $\mathcal{C}$

\State $\mathcal{C} \gets [\,]$
\State $P_{100} \gets \lfloor 100P + 0.5 \rfloor$;\;
       $M_{100} \gets \lfloor 100M + 0.5 \rfloor$;\;
       $B_{\mathrm{mis}} \gets \lfloor M L_{\max} \rfloor$
\If{$\mathit{rmq}$}
    \For{$i \leftarrow 0$ \textbf{to} $n-1$}
        \State $\mathrm{cls}[i] \gets
            \begin{cases}
                1 & \text{if } S[i]\in\{G,A\}\\
                2 & \text{if } S[i]\in\{C,T\}\\
                0 & \text{otherwise}
            \end{cases}$
    \EndFor
    \State Build $\mathrm{nextBad}[i]$, the smallest index $j \ge i$ with
           $\mathrm{cls}[j]=0$; use $n$ if no such index exists
    \State $G[0]\gets 0$;\; $C[0]\gets 0$
    \For{$i \leftarrow 0$ \textbf{to} $n-1$}
        \State $G[i+1] \gets G[i] + [\mathrm{cls}[i]=1]$
        \State $C[i+1] \gets C[i] + [\mathrm{cls}[i]=2]$
    \EndFor
    \For{$i \leftarrow 0$ \textbf{to} $n$}
        \State $F_{\mathrm{GA}}[i] \gets 100G[i] - P_{100}i$
        \State $F_{\mathrm{CT}}[i] \gets 100C[i] - P_{100}i$
    \EndFor
    \State Build range-maximum structures $T_{\mathrm{GA}}$ over $F_{\mathrm{GA}}$
           and $T_{\mathrm{CT}}$ over $F_{\mathrm{CT}}$
\EndIf

\For{each center position $c \in [L_{\min}-1,\; n-L_{\min}-1]$}
    \If{$S[c]$ is not ACGT} \Comment{skip invalid centers}
        \State \textbf{continue}
    \EndIf
    \State $s_{\max} \gets \min\bigl(S_{\max},\; n - L_{\min} - c - 1\bigr)$
    \For{each spacer $s \in [0, s_{\max}]$}
        \State $r_0 \gets c + s + 1$ \Comment{right-arm start}
        \If{$\mathit{rmq}$}
            \Comment{exact prefilter: never changes the output}
            \State $K_{\max} \gets \min\bigl(L_{\max},\; c+1,\; n-r_0,\;
                   \mathrm{nextBad}[r_0]-r_0\bigr)$
            \If{$K_{\max} < L_{\min}$}
                \State \textbf{continue}
            \EndIf
            \State $\mathrm{gaPossible} \gets
                   \textsc{RangeMax}(T_{\mathrm{GA}},\, r_0+L_{\min},\, r_0+K_{\max})
                   \ge F_{\mathrm{GA}}[r_0]$
            \State $\mathrm{ctPossible} \gets
                   \textsc{RangeMax}(T_{\mathrm{CT}},\, r_0+L_{\min},\, r_0+K_{\max})
                   \ge F_{\mathrm{CT}}[r_0]$
            \If{\textbf{not} $(\mathrm{gaPossible} \lor \mathrm{ctPossible})$}
                \State \textbf{continue} \Comment{no right-arm prefix can satisfy purity}
            \EndIf
        \EndIf
        \State $\ell \gets c$ \Comment{left pointer}
        \State $r \gets r_0$ \Comment{right pointer}
        \State $k \gets 0,\; \textit{mis} \gets 0,\; \textit{ga} \gets 0,\; \textit{ct} \gets 0$
        \State $\textit{best}_k \gets 0$
        \While{$\ell \ge 0$ \textbf{and} $r < n$ \textbf{and} $k < L_{\max}$}
            \If{$S[r]$ is not ACGT} \State \textbf{break} \EndIf
            \If{$S[\ell] \neq S[r]$} \State $\textit{mis} \gets \textit{mis} + 1$ \EndIf
            \State $k \gets k + 1$
            \If{$S[r]$ is G or A} $\textit{ga} \gets \textit{ga} + 1$ \Else $\textit{ct} \gets \textit{ct} + 1$ \EndIf
            \If{$k \ge L_{\min}$}
                \State $\textit{pure}_{\text{GA}} \gets 100\textit{ga} \ge P_{100} \cdot k$
                \State $\textit{pure}_{\text{CT}} \gets 100\textit{ct} \ge P_{100} \cdot k$
                \State $\textit{mirrorOk} \gets 100\textit{mis} \le M_{100} \cdot k$
                \If{$(\textit{pure}_{\text{GA}} \lor \textit{pure}_{\text{CT}}) \land \textit{mirrorOk}$}
                    \State $\textit{best}_k \gets k$ \Comment{record this arm length}
                \EndIf
                \If{$\textit{mis} > B_{\mathrm{mis}}$} \State \textbf{break} \EndIf
            \EndIf
            \State $\ell \gets \ell - 1,\; r \gets r + 1$
        \EndWhile
        \If{$\textit{best}_k \ge L_{\min}$}
            \State Record candidate with $\textit{arm} = \textit{best}_k$, $\textit{spacer} = s$,
                   left-arm start $c - \textit{best}_k + 1$, right-arm start $r_0$
                   \Comment*{add to $\mathcal{C}$}
        \EndIf
    \EndFor
\EndFor

\Statex \textit{$\triangleright$ Inward extension: absorb matching spacer-end pairs into the arms}
\For{each candidate $h \in \mathcal{C}$ with arm $a$, spacer $\ell_s$,
     left-arm start $\ell_0$, right-arm start $r_0$}
    \If{$\ell_s < 2$ \textbf{or} $a \ge L_{\max}$} \State \textbf{continue} \EndIf
    \State $\textit{ga} \gets \bigl|\{k<a : S[r_0{+}k]\in\{G,A\}\}\bigr|$;\;
           $\textit{ct} \gets a - \textit{ga}$
    \While{$\ell_s \ge 2$ \textbf{and} $a < L_{\max}$}
        \State $x \gets S[\ell_0 + a]$;\; $y \gets S[r_0 - 1]$
            \Comment{outermost spacer bases, left and right}
        \If{$x \neq y$ \textbf{or} $y$ is not ACGT} \State \textbf{break} \EndIf
        \State $\textit{ga}' \gets \textit{ga} + [y\in\{G,A\}]$;\;
               $\textit{ct}' \gets \textit{ct} + [y\in\{C,T\}]$
        \If{$100\,\textit{ga}' < P_{100}(a{+}1)$ \textbf{and}
            $100\,\textit{ct}' < P_{100}(a{+}1)$}
            \State \textbf{break} \Comment{extended arm would violate purity}
        \EndIf
        \State $a \gets a + 1$;\; $\ell_s \gets \ell_s - 2$;\;
               $r_0 \gets r_0 - 1$;\; $(\textit{ga}, \textit{ct}) \gets (\textit{ga}', \textit{ct}')$
    \EndWhile
    \State Update $h$ with arm $a$, spacer $\ell_s$, and purity and mirror
           identity recomputed over the new arm
           \Comment{start and end are unchanged; the mismatch count is unchanged}
\EndFor
\State Sort $\mathcal{C}$ by (start, end, arm) and keep one candidate per
       distinct triple \Comment{several $(c,s)$ may normalize to the same partition}
\State \Return $\mathcal{C}$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}
\caption{HSeeker Phase 2: Overlap Removal}
\label{alg:mirror-phase2}
\small
\begin{algorithmic}[1]
\Require Candidate list $\mathcal{C}$ from Algorithm~\ref{alg:mirror-phase1},
         strategy $\sigma \in \{\textsc{Greedy}, \textsc{Score}\}$;
         for $\sigma = \textsc{Score}$, every $h \in \mathcal{C}$ carries a
         total score $h.\textit{score}$ from Algorithm~\ref{alg:mirror-phase3}
\Ensure Non-overlapping H-DNA mirror repeat hits $\mathcal{H}$
\Statex \textit{Coordinates are 1-based and inclusive; hits on different sequences never overlap.}

\If{$\sigma = \textsc{Greedy}$}
    \State Sort $\mathcal{C}$ by start coordinate (longest arm first on ties;
           shorter spacer first among equal-arm ties)
    \State $\mathcal{H} \gets [\,]$
    \For{each candidate $h \in \mathcal{C}$}
        \If{$\mathcal{H}$ is empty \textbf{or} $h.\textit{start} > \textsc{Last}(\mathcal{H}).\textit{end}$}
            \State Append $h$ to $\mathcal{H}$ \Comment{no overlap---keep}
        \ElsIf{$h.\textit{arm} > \textsc{Last}(\mathcal{H}).\textit{arm}$ \textbf{or}
              $(h.\textit{arm} = \textsc{Last}(\mathcal{H}).\textit{arm} \land h.\textit{spacer} < \textsc{Last}(\mathcal{H}).\textit{spacer})$}
            \State Replace \textsc{Last}$(\mathcal{H})$ with $h$ \Comment{overlap but better---replace}
        \EndIf
    \EndFor
\Else
    \State Sort $\mathcal{C}$ by descending $h.\textit{score}$; ties by longer arm,
           then shorter spacer, then earlier start;
           unscored hits ($\bot$) last
    \State $\mathcal{H} \gets [\,]$
    \For{each candidate $h \in \mathcal{C}$ in this order}
        \If{$h$ overlaps no hit already in $\mathcal{H}$}
            \State Append $h$ to $\mathcal{H}$
        \EndIf
    \EndFor
    \State Sort $\mathcal{H}$ by (start, end)
\EndIf
\State \Return $\mathcal{H}$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}
\caption{HSeeker Phase 3: Thermodynamic Stability Scoring}
\label{alg:mirror-phase3}
\scriptsize
\begin{algorithmic}[1]
\Require Full motif $s = s_1 \cdot s_3 \cdot s_2$ (left arm $s_1$, spacer $s_3$,
         right arm $s_2$, each arm of length $a$);
         scoring constants: pairing energies $w_G$, $w_A$, stacking bonus $\delta$,
         penalty divisor $d$, penalty-run cap $n_{\max}$,
         minimum window $L_{\mathrm{win}}$, spacer-to-arm ratio bound $v$
\Ensure Stacking score, pairing score, total score, putative triplex
        $s_l[s_{\mathrm{spacer}}]s_r$, or $\bot$ if the arms contain a non-ACGT base

\Statex \textit{$\triangleright$ Orientation and alignment}
\State $s_1 \leftarrow s[{:}a]$;\; $s_3 \leftarrow s[a{:}-a]$;\; $s_2 \leftarrow s[-a{:}]$
\If{$|CT(s_1)| > |GA(s_1)|$}
    \State $s \leftarrow \mathrm{revcomp}(s)$; recompute $s_1, s_3, s_2$
    \Comment{score on the purine strand}
\EndIf
\State $s_2 \leftarrow \mathrm{reverse}(s_2)$
    \Comment{now $s_1[j]$ pairs with $s_2[j]$}
\If{$s_1 \cdot s_2$ contains a non-ACGT base} \State \Return $\bot$ \EndIf
\State $\ell_s \leftarrow |s_3|$

\Statex \textit{$\triangleright$ Per-position pairing score and match indicator}
\For{$j \leftarrow 0$ \textbf{to} $a - 1$}
    \State $P[j] \leftarrow
        \begin{cases}
            w_G   & \text{if } s_1[j] = s_2[j] = G\\
            w_A   & \text{if } s_1[j] = s_2[j] = A\\
            -(w_G + w_A)/2 & \text{otherwise}
        \end{cases}$
    \State $B[j] \leftarrow 1$ if $s_1[j] = s_2[j] \in \{G,A\}$ else $0$
\EndFor

\Statex \textit{$\triangleright$ Stacking score of a window $B[L{:}R]$}
\Procedure{CalcStacking}{$B[L{:}R]$}
    \State $K[0] \leftarrow 0$;\; $c \leftarrow 0$
        \Comment{$c$: current run of non-stacked positions}
    \For{$j \leftarrow 1$ \textbf{to} $R - L - 1$}
        \If{$B[L{+}j{-}1] = B[L{+}j] = 1$}
            \State $K[j] \leftarrow +\delta$;\; $c \leftarrow 0$
        \Else
            \State $n \leftarrow \min(\max(c - 1,\, 0),\, n_{\max})$
            \State $K[j] \leftarrow -\bigl(\delta \text{ if } n = 0 \text{ else }
                   \delta^{\,n+1}/d\bigr)$;\; $c \leftarrow c + 1$
        \EndIf
    \EndFor
    \State \Return $K$
\EndProcedure

\Statex \textit{$\triangleright$ Optimal arm window $[L, R)$ under the spacer constraint}
\State $\mathit{best} \leftarrow -\infty$;\; $(L^*, R^*) \leftarrow \mathrm{None}$
\For{$L \leftarrow 0$ \textbf{to} $a - 1$}
    \For{$R \leftarrow L + \min(L_{\mathrm{win}},\, a)$ \textbf{to} $a$}
        \If{$2(a - R) + \ell_s \;\le\; (R - L)\cdot v$}
            \Comment{new spacer length $\le$ window length $\times\, v$}
            \State $K \leftarrow \textsc{CalcStacking}(B[L{:}R])$
            \State $\mathit{cur} \leftarrow
                   \mathrm{round}\!\bigl(\textstyle\sum_{j=L}^{R-1} P[j]
                   + \sum_j K[j],\; 3\bigr)$
            \If{$\mathit{cur} > \mathit{best}$}
                \Comment{strict: first maximum in $(L,R)$ order wins}
                \State $\mathit{best} \leftarrow \mathit{cur}$;\;
                       $(L^*, R^*) \leftarrow (L, R)$
            \EndIf
        \EndIf
    \EndFor
\EndFor
\If{$(L^*, R^*) = \mathrm{None}$}
    \State $(L^*, R^*) \leftarrow (0, a)$
    \Comment{no window fits: score the detected arms as they are}
\EndIf

\Statex \textit{$\triangleright$ Final scores and putative triplex}
\State $K \leftarrow \textsc{CalcStacking}(B[L^*{:}R^*])$
\State $\mathit{stacking\_score} \leftarrow \textstyle\sum_j K[j]$;\;
       $\mathit{pairing\_score} \leftarrow \textstyle\sum_{j=L^*}^{R^*-1} P[j]$
\State $\mathit{total\_score} \leftarrow
       \max\bigl(\mathrm{round}(\mathit{stacking\_score} + \mathit{pairing\_score},\; 3),\; 0\bigr)$
\State $s_{\mathrm{spacer}} \leftarrow
       s_1[R^*{:}] \;+\; s_3 \;+\; \mathrm{reverse}(s_2[R^*{:}])$
    \Comment{right tail of $s_1$ + spacer + left tail of the original right arm}
\State $s_l \leftarrow s_1[L^*{:}R^*]$;\;
       $s_r \leftarrow \mathrm{reverse}(s_2[L^*{:}R^*])$
\State $\mathit{putative\_triplex} \leftarrow
       s_l + \text{``[''} + s_{\mathrm{spacer}} + \text{``]''} + s_r$
\State \Return $\{\mathit{stacking\_score},\;
                  \mathit{pairing\_score},\;
                  \mathit{total\_score},\;
                  \mathit{putative\_triplex}\}$
\end{algorithmic}
\end{algorithm}
